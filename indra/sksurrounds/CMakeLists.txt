# -*- cmake -*-

# The distant regions' layout, free of the viewer: the restricted quadtree
# of far nodes around the live regions and their ring, the edge levels
# their seams share, and the map tile each node reads.

set(sksurrounds_SOURCES sksurroundsfield.cpp)

set(sksurrounds_HEADERS sksurroundsfield.h)

add_library(sksurrounds STATIC)
target_sources(
  sksurrounds
  PRIVATE ${sksurrounds_SOURCES}
  PUBLIC FILE_SET HEADERS FILES ${sksurrounds_HEADERS}
)

target_link_libraries(sksurrounds PRIVATE al::flags PUBLIC llcommon)

if(AL_BUILD_TESTS)
  # Selection and edge levels against the oracle's golden cameras
  # (tests/sksurroundsfixture.inl, generated by the oracle's gen_fixture.py).
  al_add_test_binary(
    sksurrounds_tests
    PROJECT sksurrounds
    SOURCES tests/sksurroundsfield_test.cpp
    LIBRARIES sksurrounds
    SUPPORT llcommon_test_checks
  )
endif()
